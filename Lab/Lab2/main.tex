% ==================== BÀI 2.1 ====================
\subsection{Bài 2.1}

\textbf{Giải thích:}
Ta tính độ dài chuỗi \texttt{msg}, lưu ở biến \texttt{msg\_len}. Cộng thêm \texttt{0x80} để chuyển sang ASCII. Khi khởi chạy, chương trình chỉ cần nạp đúng địa chỉ buffer và output chuỗi.

\textbf{Mã nguồn chương trình (\texttt{bai2.1.s}):}
\begin{lstlisting}
.section .data
msg:
	.string "IloveCR7"
msg_len = . - msg - 1 # Lay dia chi hien tai - dia chi msg - 1 (tru byte null)

output:
	.byte msg_len + 0x30 # Cong 48 (0x30) de chuyen thanh ky tu ASCII
	.byte 0x0a           # Ky tu xuong dong '\n'

.section .text
	.globl _start
_start:
	movl $output, %ecx   # Dia chi buffer can xuat
	movl $0x2, %edx      # Do dai xuat ra: 2 bytes
	movl $0x1, %ebx      # File descriptor: stdout (1)
	movl $0x4, %eax      # System call number: sys_write (4)
	int $0x80            # Goi kernel xuat chuoi

	movl $0x1, %eax      # System call number: sys_exit (1)
	int $0x80            # Thoat chuong trinh
\end{lstlisting}

% ==================== BÀI 2.2 ====================
\subsection{Bài 2.2}

\textbf{Giải thích:}
Nhập lần lượt 4 số. Với mỗi số, ta nhận kí tự hàng chục, chuyển sang số nguyên, nhân 10; nhận kí tự hàng đơn vị, chuyển sang số nguyên; lưu tổng lại ta được số tương ứng. Sau đó, ta cộng 4 số đã được nhập vào rồi tiến hành nạp tham số vào các thanh ghi để chia cho 8 lấy số dư, chuyển kết quả sang mã ASCII rồi in ra màn hình.

\textbf{Mã nguồn chương trình (\texttt{bai2\_2.s}):}
\begin{lstlisting}
.section .data
nhapvao:
	.string  "Enter a number (2-digit): "
kq:
	.string "Remainder: "
newline:
	.byte 10
ans:
	.int 0

.section .bss
	.lcomm number1, 4

.section .text
	.globl _start

jmp _start

nhapso:
	movl $4, %eax        # sys_write in loi nhac
	movl $1, %ebx
	movl $nhapvao, %ecx
	movl $26, %edx
	int $0x80

	movl $3, %eax        # sys_read doc 3 bytes tu STDIN
	movl $0, %ebx
	movl $number1, %ecx
	movl $3, %edx
	int $0x80

	movzbl number1, %eax    # Doc chu so hang chuc
	subl $48, %eax          # Chuyen thanh gia tri nguyen
	imull $10, %eax         # Nhan voi 10

	movzbl number1+1, %ebx  # Doc chu so hang don vi
	subl $48, %ebx
	addl %ebx, %eax         # Tong gia tri cua so vua nhap

	addl %eax, ans          # Cong don vao bien tong ans
	ret

inkq:
	movl ans, %eax       # Nap tong vao %eax
	xorl %edx, %edx      # Xoa %edx de chuan bi chia 64-bit
	movl $8, %ecx        # So chia = 8
	divl %ecx            # Chia: thuong o %eax, so du o %edx

	movl %edx, ans
	addl $48, ans        # Chuyen so du sang ky tu ASCII

	movl $4, %eax        # In chuoi thong bao "Remainder: "
	movl $1, %ebx
	movl $kq, %ecx
	movl $11, %edx
	int $0x80

	movl $4, %eax        # In ky tu so du
	movl $1, %ebx
	movl $ans, %ecx
	movl $1, %edx
	int $0x80

	movl $4, %eax        # In dau xuong dong
	movl $1, %ebx
	movl $newline, %ecx
	movl $1, %edx
	int $0x80
	ret

_start:
	call nhapso
	call nhapso
	call nhapso
	call nhapso
	call inkq

	mov $1, %eax         # sys_exit
	int $0x80
\end{lstlisting}

% ==================== BÀI 2.3 ====================
\subsection{Bài 2.3}

\textbf{Giải thích:}
Ta nhập chuỗi và số bước dịch $n$ (từ 0 đến 9). Chương trình duyệt qua từng ký tự: nếu là chữ in hoa hoặc chữ in thường, ta dịch đi $n$ đơn vị và kiểm tra xoay vòng bằng cách trừ đi 26 nếu vượt quá bảng chữ cái; các ký tự còn lại (số, khoảng trắng, ký tự đặc biệt) được giữ nguyên. Cuối cùng, ta xuất chuỗi đã được mã hóa ra màn hình.


\textbf{Mã nguồn chương trình (\texttt{bai2.3.s}):}
\begin{lstlisting}
.section .rodata
enter_string:
    .ascii "Enter a string (<255 chars): "
enter_string_len = .-enter_string

shift_string:
    .ascii "Enter n (0-9): "
shift_string_len = .-shift_string

.section .bss
    .lcomm input_str, 256
    .lcomm input_len, 4
    .lcomm n, 2

.section .text
	.globl _start

printf:
	movl $0x1, %ebx         # stdout
	movl $0x4, %eax         # write
	int $0x80
	ret

read:
	xorl %ebx, %ebx         # stdin
	movl $0x3, %eax         # read
	int $0x80
	ret

cipher:
	movl $0, %ecx
	movl $input_str, %edx
	decl %edx

.cipher_loop:
	cmpl input_len, %ecx
	je .fin
	incl %edx
	incl %ecx
	cmpb $'A', (%edx)
	jb .cipher_loop     # Ky tu ngoai khoang (giu nguyen)
	cmpb $'Z', (%edx)
	movb $'A', %bl
	jbe .shift          # Chu hoa ('A' <= x <= 'Z')

	cmpb $'a', (%edx)
	jb .cipher_loop     # Ky tu dac biet nam giua 'Z' va 'a'
	cmpb $'z', (%edx)
	movb $'a', %bl
	jbe .shift          # Chu thuong ('a' <= x <= 'z')

	jmp .cipher_loop    # Ky tu lon hon 'z'

.fin:
	ret

.shift:
	movb (%edx), %al
	subb %bl, %al       # Dua ve khoang 0 .. 25
	addb n, %al         # Cong buoc dich n
	cmpb $26, %al       # Kiem tra tran 26
	jb .done
	subb $26, %al       # Xoay vong bang cach tru 26

.done:
	addb %bl, %al       # Dua ve lai ma ASCII goc
	movb %al, (%edx)    # Ghi de ky tu da ma hoa vao buffer
	jmp .cipher_loop

_start:
	movl $enter_string, %ecx
	movl $enter_string_len, %edx
	call printf

	movl $input_str, %ecx
	movl $256, %edx
	call read
	movl %eax, input_len

	movl $shift_string, %ecx
	movl $shift_string_len, %edx
	call printf

	movl $n, %ecx
	movl $0x2, %edx
	call read
	subb $0x30, n
	call cipher

	movl $input_str, %ecx
	movl input_len, %edx
	call printf

	movl $1, %eax
	int $0x80
\end{lstlisting}

% ==================== BÀI 2.4 ====================
\subsection{Bài 2.4}

\textbf{Giải thích:}
Thay vì duyệt từng số từ $x$ đến $y$, ta áp dụng công thức $\lfloor y/10 \rfloor - \lfloor (x-1)/10 \rfloor$ để đếm trực tiếp số lượng bội của 10 trong đoạn $[x, y]$. Sau khi nhập hai số $x$ và $y$, chương trình tính thương số tương ứng của từng số khi chia cho 10 rồi lấy hiệu của chúng, chuyển kết quả sang mã ASCII và đẩy vào ngăn xếp để in ra màn hình.


\textbf{Mã nguồn chương trình (\texttt{bai2.4.s}):}
\begin{lstlisting}
.section .data
hihi:
	.string "Nhap so: "
string_len = . - hihi
newline:
	.ascii "\n"

.section .bss
	.lcomm x, 3
	.lcomm y, 3

.section .text
.globl _start

_start:
	# 1. Nhap va tinh [(x - 1) / 10]
	movl $4, %eax
	movl $1, %ebx
	movl $hihi, %ecx
	movl $string_len, %edx
	int $0x80
	
	movl $0, %ebx
	movl $3, %eax
	movl $x, %ecx
	movl $3, %edx
	int $0x80
	
	movzbl x, %edi
	subl $'0', %edi
	imull $10, %edi
	
	movzbl x+1, %eax
	subl $'0', %eax
	addl %eax, %edi
	
	decl %edi            # Tinh x - 1
	movl $10, %ecx
	movl %edi, %eax
	xorl %edx, %edx
	divl %ecx            # Chia 10: thuong nam o %eax
	movl %eax, %edi      # Luu [(x - 1) / 10] vao %edi

	# 2. Nhap va tinh [y / 10]
	movl $4, %eax
	movl $1, %ebx
	movl $hihi, %ecx
	movl $string_len, %edx 
	int $0x80

	movl $0, %ebx
	movl $3, %eax
	movl $y, %ecx
	movl $3, %edx
	int $0x80

	movzwl y, %eax
	subb $'0', %al
	subb $'0', %ah

	movzbl %al, %ecx
	imull $10, %ecx
	movzbl %ah, %edx
	addl %edx, %ecx      # y gio nam o %ecx
	
	movl %ecx, %eax	
	movl $10, %ecx
	xorl %edx, %edx
	divl %ecx            # Chia 10: thuong o %eax
	movl %eax, %esi      # Luu [y / 10] vao %esi

	# 3. Tinh hieu so va xuat ket qua
	subl %edi, %esi      # %esi = [y/10] - [(x-1)/10]
	addl $'0', %esi      # Chuyen thanh ky tu ASCII
	pushl %esi           # Day vao dinh ngan xep
	
	movl $4, %eax
	movl $1, %ebx
	movl %esp, %ecx      # Dung dia chi %esp lam buffer xuat
	movl $1, %edx
	int $0x80

	movl $4, %eax        # Xuat dau xuong dong
	movl $1, %ebx
	movl $newline, %ecx
	movl $1, %edx
	int $0x80

	movl $1, %eax        # Thoat
	xorl %ebx, %ebx
	int $0x80
\end{lstlisting}

% ==================== BÀI 2.5 ====================
\subsection{Bài 2.5}

\textbf{Giải thích:}
Ta nhập năm sinh 4 chữ số, chuyển thành số nguyên rồi lấy mốc năm 2026 trừ đi để tính tuổi. Nếu tuổi nằm trong khoảng từ 16 đến 30 thì in ra \texttt{"Thuoc Doan TN"}, ngược lại in \texttt{"Khong thuoc Doan TN"}. Tiếp theo, ta chia liên tiếp số tuổi cho 10 rồi dùng ngăn xếp (stack) để in ra số tuổi cụ thể theo đúng thứ tự các chữ số.


\textbf{Mã nguồn chương trình (\texttt{bai2.5.s}):}
\begin{lstlisting}
.section .data
hello:
	.ascii "Nhap nam sinh: "
string_len = . - hello

Thuoc:
	.ascii "Thuoc Doan TN\n"
Thuoc_len = . - Thuoc

Khongthuoc:
	.ascii "Khong thuoc Doan TN\n"
Khongthuoc_len = . - Khongthuoc

intuoi:
	.ascii "Tuoi: "
intuoi_len = . - intuoi

newline:
	.ascii "\n"

.section .bss
	.lcomm tuoi, 5
	.lcomm out, 1

.section .text
.globl _start

_start:
	# 1. Nhap nam sinh tu ban phim
	movl $4, %eax
	movl $1, %ebx
	movl $hello, %ecx
	movl $string_len, %edx
	int $0x80

	movl $3, %eax
	movl $0, %ebx
	movl $tuoi, %ecx
	movl $5, %edx
	int $0x80

	# 2. Chuyen 4 ky tu thanh so nguyen trong %edi
	xorl %edi, %edi
		
	movzbl tuoi, %eax
	subl $'0', %eax
	imull $1000, %eax
	addl %eax, %edi
	
	movzbl tuoi+1, %eax
	subl $'0', %eax
	imull $100, %eax 
	addl %eax, %edi

	movzbl tuoi+2, %eax
	subl $'0', %eax
	imull $10, %eax 
	addl %eax, %edi

	movzbl tuoi+3, %eax
	subl $'0', %eax 
	addl %eax, %edi

	# 3. Tinh tuoi: 2026 - nam_sinh
	movl $2026, %eax
	subl %edi, %eax
	movl %eax, %edi
	
	# 4. Kiem tra khoang [16, 30]
	cmpl $16, %edi
	jl ngoai_khoang
	cmpl $30, %edi
	jg ngoai_khoang
	
	movl $4, %eax
	movl $1, %ebx
	movl $Thuoc, %ecx
	movl $Thuoc_len, %edx
	int $0x80
	jmp nang_cao

ngoai_khoang:
	movl $4, %eax
	movl $1, %ebx
	movl $Khongthuoc, %ecx
	movl $Khongthuoc_len, %edx
	int $0x80

nang_cao:
	movl $4, %eax
	movl $1, %ebx
	movl $intuoi, %ecx
	movl $intuoi_len, %edx
	int $0x80
	call print

done:
	movl $1, %eax
	xorl %ebx, %ebx
	int  $0x80

# Thu tuc print: in so nguyen trong %edi ra man hinh
print:
	movl %edi, %eax
	xorl %esi, %esi

chia_loop:
	xorl %edx, %edx
	movl $10, %ebx
	divl %ebx            # Chia 10 lay so du vao %edx
	addb $'0', %dl       # Chuyen sang ma ASCII
	pushl %edx           # Day vao ngan xep
	incl %esi            # Tang bien dem chu so
	cmpl $0, %eax
	jne chia_loop

in_loop:
	popl %eax            # Rut chu so theo thu tu dung
	movb %al, out
	movl $4, %eax
	movl $1, %ebx
	movl $out, %ecx
	movl $1, %edx
	int $0x80
	decl %esi
	jnz in_loop

	movl $4, %eax        # In ky tu xuong dong
	movl $1, %ebx
	movl $newline, %ecx
	movl $1, %edx
	int $0x80
	ret
\end{lstlisting}

% ==================== BÀI 2.6 ====================
\subsection{Bài 2.6}

\textbf{Giải thích:}
Ta nhập năm 4 chữ số và kiểm tra năm nhuận theo lịch Gregory: nếu là năm tròn thế kỷ (kết thúc bằng \texttt{00}) thì kiểm tra chia hết cho 400; các năm còn lại chỉ cần chia hết cho 4. Sau khi in kết quả \texttt{"Nam nhuan"} hoặc \texttt{"Nam khong nhuan"}, chương trình liên tục tăng năm lên 1 và kiểm tra cho đến khi tìm được năm nhuận kế tiếp rồi in ra màn hình.


\textbf{Mã nguồn chương trình (\texttt{bai2\_6.s}):}
\begin{lstlisting}
.section .data
nhapvao:
	.string  "Enter a year (4-digit): "
kqnamnhuan:
	.string "Nam nhuan\n"
kqnamkhongnhuan:
	.string "Nam khong nhuan\n"
namnhuantieptheouwu:
	.string "Nam nhuan ke tiep: "
newline:
	.byte 10
ans:
	.int 0

.section .bss
	.lcomm number1, 5
	buffer_num: .space 16

.section .text
	.globl _start

jmp _start

namnhuantieptheo:
	addl $1, ans         # Tang nam len 1
	movl ans, %eax
	xorl %edx, %edx
	movl $4, %ecx
	divl %ecx
	cmpl $0, %edx 
	jne namnhuantieptheo # Khong chia het cho 4 -> lap tiep
    
	movl ans, %eax
	xorl %edx, %edx
	movl $100, %ecx
	divl %ecx
	cmpl $0, %edx
	jne okroido          # Chia het cho 4 va khong chia het cho 100 -> Nam nhuan!

	movl ans, %eax
	xorl %edx, %edx
	movl $400, %ecx
	divl %ecx
	cmpl $0, %edx
